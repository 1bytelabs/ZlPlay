cmake_minimum_required(VERSION 3.24)
project(ZLPlayMobile VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(ZLPLAY_JUCE_SOURCE "" CACHE PATH "Existing JUCE 9.0.4 checkout")
option(ZLPLAY_HOST_CHECK "Build embedded engine regression on the host" OFF)
if(ZLPLAY_JUCE_SOURCE)
  add_subdirectory("${ZLPLAY_JUCE_SOURCE}" "${CMAKE_BINARY_DIR}/juce")
  set(zlplay_juce_dir "${ZLPLAY_JUCE_SOURCE}")
else()
  include(FetchContent)
  FetchContent_Declare(JUCE URL https://codeload.github.com/juce-framework/JUCE/tar.gz/3fb5cbc0340758596dac39f7c8de47d562b4896e URL_HASH SHA256=2bb7095497a7bb7269dbdf841a7c6aa0939eaeb2a2bc9724ac4758d3ae3565a7 DOWNLOAD_EXTRACT_TIMESTAMP FALSE)
  FetchContent_MakeAvailable(JUCE)
  set(zlplay_juce_dir "${juce_SOURCE_DIR}")
endif()
if(ZLPLAY_HOST_CHECK)
  juce_add_console_app(ZLPlayEmbeddedCheck PRODUCT_NAME "ZLPlayEmbeddedCheck")
  set(app ZLPlayEmbeddedCheck)
  target_sources(${app} PRIVATE Tests/EmbeddedCheck.cpp Source/ProjectImport.cpp Source/RemoteTransport.mm)
  target_link_libraries(${app} PRIVATE "-framework Security")
else()
  juce_add_gui_app(ZLPlayMobile PRODUCT_NAME "ZLPlay Mobile" COMPANY_NAME "1Byte Labs"
    BUNDLE_ID com.onebytelabs.zlplay.ios.dev TARGETED_DEVICE_FAMILY "1,2"
    ICON_BIG ../desktop/icon.png FILE_SHARING_ENABLED TRUE REQUIRES_FULL_SCREEN TRUE
    PLIST_TO_MERGE "<plist><dict><key>NSCameraUsageDescription</key><string>Scan the ZLPlay pairing QR code on your computer.</string><key>NSLocalNetworkUsageDescription</key><string>Connect to ZLPlay on your computer over the local network.</string><key>LSSupportsOpeningDocumentsInPlace</key><true/><key>UIStatusBarStyle</key><string>UIStatusBarStyleLightContent</string></dict></plist>"
    IPHONE_SCREEN_ORIENTATIONS UIInterfaceOrientationLandscapeLeft UIInterfaceOrientationLandscapeRight
    IPAD_SCREEN_ORIENTATIONS UIInterfaceOrientationLandscapeLeft UIInterfaceOrientationLandscapeRight)
  foreach(locale en pt-BR es)
    target_sources(ZLPlayMobile PRIVATE "Resources/${locale}.lproj/InfoPlist.strings")
    set_source_files_properties("Resources/${locale}.lproj/InfoPlist.strings" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/${locale}.lproj")
  endforeach()
  juce_add_binary_data(ZLPlayMobileBrand SOURCES ../desktop/icon.png)
  set(app ZLPlayMobile)
  target_link_libraries(${app} PRIVATE ZLPlayMobileBrand)
  target_sources(${app} PRIVATE Source/Main.cpp Source/ProjectImport.cpp)
  target_link_libraries(${app} PRIVATE juce::juce_gui_extra)
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_sources(${app} PRIVATE Source/PlatformInsets.mm Source/MobileRemote.mm Source/RemoteTransport.mm)
    target_link_libraries(${app} PRIVATE "-framework AVKit" "-framework Security")
    set_target_properties(${app} PROPERTIES XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET "15.0")
  endif()
  target_sources(${app} PRIVATE ../app/Meuclick.wav)
  set_source_files_properties(../app/Meuclick.wav PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set(legal_dir "${CMAKE_CURRENT_BINARY_DIR}/legal")
  file(MAKE_DIRECTORY "${legal_dir}")
  configure_file(../LICENSE "${legal_dir}/ZLPlay-AGPL-3.0.txt" COPYONLY)
  configure_file(../THIRD_PARTY_NOTICES.md "${legal_dir}/THIRD_PARTY_NOTICES.md" COPYONLY)
  configure_file("${zlplay_juce_dir}/LICENSE.md" "${legal_dir}/JUCE-LICENSE.md" COPYONLY)
  configure_file(../native/vendor/signalsmith-stretch-main/LICENSE.txt "${legal_dir}/Signalsmith-Stretch-LICENSE.txt" COPYONLY)
  configure_file(../native/vendor/linear-main/LICENSE.txt "${legal_dir}/Signalsmith-Linear-LICENSE.txt" COPYONLY)
  configure_file(../native/vendor/libebur128/COPYING "${legal_dir}/libebur128-LICENSE.txt" COPYONLY)
  foreach(name ZLPlay-AGPL-3.0.txt THIRD_PARTY_NOTICES.md JUCE-LICENSE.md Signalsmith-Stretch-LICENSE.txt Signalsmith-Linear-LICENSE.txt libebur128-LICENSE.txt)
    target_sources(${app} PRIVATE "${legal_dir}/${name}")
    set_source_files_properties("${legal_dir}/${name}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources/licenses)
  endforeach()
endif()
target_sources(${app} PRIVATE ../native/engine.cpp ../native/vendor/libebur128/ebur128.c)
target_include_directories(${app} PRIVATE ../native ../native/vendor/libebur128 ../native/vendor/signalsmith-stretch-main ../native/vendor/linear-main/include)
target_compile_definitions(${app} PRIVATE ZLPLAY_EMBEDDED_ENGINE=1 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_STANDALONE_APPLICATION=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_MP3AUDIOFORMAT=1 JUCE_MODAL_LOOPS_PERMITTED=0)
target_link_libraries(${app} PRIVATE juce::juce_audio_devices juce::juce_audio_formats juce::juce_recommended_config_flags)

add_custom_command(TARGET ${app} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/../app/Meuclick.wav" "$<TARGET_FILE_DIR:${app}>/Meuclick.wav")
